### Step 3: Create repositories
Create api_server/db/repositories/department.py:
```python
from typing import List, Optional
from datetime import datetime
from ..models.department import Department
from .base import BaseRepository
import uuid

class DepartmentRepository(BaseRepository):
    async def get_all(self) -> List[Department]:
        result = await self.db.execute(select(Department))
        return result.scalars().all()

    async def get_by_id(self, department_id: str) -> Optional[Department]:
        result = await self.db.execute(select(Department).where(Department.id == department_id))
        return result.scalar_one_or_none()

    async def create(self, **data) -> Department:
        dept = Department(id=str(uuid.uuid4()), **data)
        self.db.add(dept)
        await self.db.commit()
        await self.db.refresh(dept)
        return dept

    async def update(self, department_id: str, **data) -> Optional[Department]:
        dept = await self.get_by_id(department_id)
        if not dept:
            return None
        for key, value in data.items():
            setattr(dept, key, value)
        dept.updated_at = datetime.utcnow()
        await self.db.commit()
        await self.db.refresh(dept)
        return dept

    async def delete(self, department_id: str) -> bool:
        dept = await self.get_by_id(department_id)
        if not dept:
            return False
        await self.db.delete(dept)
        await self.db.commit()
        return True
```
